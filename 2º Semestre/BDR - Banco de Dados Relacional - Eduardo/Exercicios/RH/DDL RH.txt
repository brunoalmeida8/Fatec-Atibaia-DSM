CREATE DATABASE RH;
USE RH;

-- Create table Estados Federação
CREATE TABLE ESTADOS(
    ID_ESTADO INT AUTO_INCREMENT PRIMARY KEY, 
    SIGLA CHAR(2) NOT NULL,
    DS_ESTADO VARCHAR(50) NOT NULL UNIQUE
);

--Create table Municipios
CREATE TABLE MUNICIPIOS(
    ID_MUNICIPIO INT AUTO_INCREMENT,
    ID_ESTADO INT NOT NULL,
    CD_IBGE INT NOT NULL,
    DS_MUNICIPIO VARCHAR(100) NOT NULL,
    PRIMARY KEY(ID_MUNICIPIO),
    
    CONSTRAINT FK_MUN_ESTADO FOREIGN KEY (ID_ESTADO)
    REFERENCES ESTADOS(ID_ESTADO)
);

--Create table Pessoas
CREATE TABLE PESSOAS(
    ID_PESSOA INT,
    ID_MUNICIPIO_NATURAL NOT NULL,
    ID_MUNICIPIO_RESIDE INT NOT NULL,
    NR_CPF CHAR(11) NOT NULL,
    NM_PESSOA VARCHAR(50) NOT NULL,

    CONSTRAINT FK_MUN_NAT_MUNICIPIO FOREIGN KEY (ID_MUNICIPIO_NATURALIDADE)
    REFERENCES MUNICIPIOS(ID_MUNICIPIO),

    CONSTRAINT FK_MUN_NASC_MUNICIPIO FOREIGN KEY (ID_MUNICIPIO_RESIDENCIA)
    REFERENCES MUNICIPIOS(ID_MUNICIPIO)
);
