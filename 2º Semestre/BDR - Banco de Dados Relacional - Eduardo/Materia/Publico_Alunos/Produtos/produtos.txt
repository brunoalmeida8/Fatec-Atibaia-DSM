Create database if not exists produtos;
use produtos;

-- Criar tabela de Unidade de Medida
Create table if not exists unidades_medidas(
  id_unidade_medida int auto_increment primary key,
  sg_unidade_medida char(2) not null,
  ds_unidade_medida varchar(40) not null unique);

-- Criar tabela de Produtos
create table if not exists produtos(
id_produto int auto_increment primary key,
ds_produto varchar(50) not null unique,
id_unidade_medida int,
-- Constraint FK com Unidade de Medida
Constraint fk_produtos_unidade 
  foreign key(id_unidade_medida)
   references unidades_medidas(id_unidade_medida)
   on delete cascade);
   -- on delete restrict
   -- on update restrict
   -- on delete set null);

/*
-- Possibilidades

ON DELETE CASCADE 
 -> Ao excluir uma unidade de medida, todos os produtos associados a ela também serão excluídos
ON DELETE RESTRICT
 -> Não permite excluir a unidade de medida enquanto existir algum produto utilizando-a.
ON DELETE NO ACTION
 -> Idem acima
ON DELETE SET NULL
 -> Ao excluir a unidade de medida, o produto continua existindo, mas id_unidade_medida passa a ser NULL. 
 -> id_unidade_medida não pode ser NOT NULL
ON UPDATE CASCADE
 -> Se o ID da unidade medida na PK for alterado, o banco atualizaria automaticamente as tabelas filhas:
ON UPDATE RESTRICT
 -> Impede a alteração em cascata da PK nas tabelas filhas. Mantém a integridade referencial
ON UPDATE NO ACTION
ON UPDATE SET NULL

*/
