/*
===============================================================================
BANCO DE DADOS: lojavirtual
SGBD: MySQL 8.0+
Objetivo: simular produtos, clientes, carrinhos e formas de pagamento.
Observação de modelagem:
- Cada registro de Carrinho representa um item/produto comprado no carrinho.
- Um mesmo código de carrinho (id_compra) pode agrupar vários produtos para
  representar uma compra com mais de um item. A tabela Carrinho possui 10 linhas,
  distribuídas em 5 compras/clientes, com dois itens por compra.
- Os valores totais da compra e do frete são repetidos nas linhas da mesma compra.
===============================================================================
*/

DROP DATABASE IF EXISTS lojavirtual;
CREATE DATABASE lojavirtual;

USE lojavirtual;

-- 1. Estados da Federação
CREATE TABLE Estados (
    id_estado INT PRIMARY KEY,
    descricao_estado VARCHAR(60) NOT NULL,
    sigla_estado CHAR(2) NOT NULL UNIQUE,
    regiao ENUM('NORTE','NORDESTE','CENTRO-OESTE','SUDESTE','SUL') NOT NULL
);

-- 2. Unidades de Medidas
CREATE TABLE Unidades_Medidas (
    id_unidade INT PRIMARY KEY,
    descricao_unidade VARCHAR(60) NOT NULL,
    sigla_unidade VARCHAR(10) NOT NULL UNIQUE
);

-- 3. Formas de Pagamento
CREATE TABLE Forma_Pagamento (
    id_forma_pagamento INT PRIMARY KEY,
    descricao_forma_pagamento VARCHAR(60) NOT NULL UNIQUE
);

-- 4. Produtos
CREATE TABLE Produtos (
    id_produto INT PRIMARY KEY,
    descricao_produto VARCHAR(120) NOT NULL,
    id_unidade INT NOT NULL,
    valor_produto DECIMAL(10,2) NOT NULL,
    valor_frete DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    status_produto ENUM('ATIVO','INATIVO') NOT NULL DEFAULT 'ATIVO',
    CONSTRAINT ck_produto_valor CHECK (valor_produto >= 0),
    CONSTRAINT ck_produto_frete CHECK (valor_frete >= 0),
    CONSTRAINT fk_produto_unidade FOREIGN KEY (id_unidade)
        REFERENCES Unidades_Medidas(id_unidade)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

-- 5. Clientes
CREATE TABLE Clientes (
    id_cliente INT PRIMARY KEY,
    nome_cliente VARCHAR(120) NOT NULL,
    cpf CHAR(11) NOT NULL UNIQUE,
    email VARCHAR(150) NOT NULL UNIQUE,
    telefone VARCHAR(20),
    data_nascimento DATE,
    logradouro VARCHAR(120) NOT NULL,
    numero VARCHAR(10) NOT NULL,
    complemento VARCHAR(60),
    bairro VARCHAR(80) NOT NULL,
    cidade VARCHAR(80) NOT NULL,
    cep CHAR(8) NOT NULL,
    id_estado INT NOT NULL,
    data_cadastro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_cliente_estado FOREIGN KEY (id_estado)
        REFERENCES Estados(id_estado)
        ON UPDATE CASCADE ON DELETE RESTRICT
);

-- 6. Carrinho: cada linha é um item; id_compra agrupa itens da mesma compra.
CREATE TABLE Carrinho (
    id_carrinho INT PRIMARY KEY,
    id_compra INT NOT NULL,
    id_cliente INT NOT NULL,
    id_produto INT NOT NULL,
    quantidade DECIMAL(10,3) NOT NULL DEFAULT 1,
    valor_unitario DECIMAL(10,2) NOT NULL,
    valor_frete_item DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    valor_total_compra DECIMAL(12,2) NOT NULL,
    valor_total_frete DECIMAL(12,2) NOT NULL,
    id_forma_pagamento INT NOT NULL,
    data_compra DATETIME NOT NULL,
    status_compra ENUM('ABERTO','PAGO','CANCELADO','ENVIADO','ENTREGUE') NOT NULL,
    CONSTRAINT ck_carrinho_quantidade CHECK (quantidade > 0),
    CONSTRAINT ck_carrinho_valores CHECK (
        valor_unitario >= 0 AND valor_frete_item >= 0
        AND valor_total_compra >= 0 AND valor_total_frete >= 0
    ),
    CONSTRAINT fk_carrinho_cliente FOREIGN KEY (id_cliente)
        REFERENCES Clientes(id_cliente)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_carrinho_produto FOREIGN KEY (id_produto)
        REFERENCES Produtos(id_produto)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_carrinho_pagamento FOREIGN KEY (id_forma_pagamento)
        REFERENCES Forma_Pagamento(id_forma_pagamento)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    INDEX idx_carrinho_compra (id_compra),
    INDEX idx_carrinho_cliente (id_cliente)
);
