select u.descricao_unidade unidade,
       p.descricao_produto produto
    from produtos p 
  join unidades_medidas u  
  on p.id_unidade = u.id_unidade


select u.descricao_unidade unidade,
       count(*) quantidade,
       sum(p.valor_produto) soma_total,
       max(p.valor_produto) maximo,
       min(p.valor_produto) minimo,
       round(avg(p.valor_produto),3) media
    from produtos p 
  join unidades_medidas u  
  on p.id_unidade = u.id_unidade
where u.descricao_unidade not in('Metro','Kit')
group by u.descricao_unidade
having media > 100 
order by u.descricao_unidade


select u.descricao_unidade unidade,
       count(*) quantidade,
       sum(p.valor_produto) soma_total,
       max(p.valor_produto) maximo,
       min(p.valor_produto) minimo,
       round(avg(p.valor_produto),3) media
    from produtos p 
  join unidades_medidas u  
  on p.id_unidade = u.id_unidade
where u.descricao_unidade not in('Metro','Kit')
group by u.descricao_unidade
having minimo > 7 and maximo < 30
order by u.descricao_unidade;
