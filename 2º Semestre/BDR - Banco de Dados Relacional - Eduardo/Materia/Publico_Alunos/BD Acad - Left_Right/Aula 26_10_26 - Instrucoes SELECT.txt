SELECT F.NM_FUNCIONARIO, F.VL_SALARIO,
       D.DS_DEPARTAMENTO
    FROM FUNCIONARIOS f
    JOIN departamentos D 
      ON F.id_departamento = D.id_departamento;

SELECT D.DS_DEPARTAMENTO DEPARTAMENTO, 
       COUNT(*) QTDE,
       SUM(F.VL_SALARIO) FOLHA
    FROM FUNCIONARIOS f
    JOIN departamentos D 
      ON F.id_departamento = D.id_departamento
GROUP BY D.DS_DEPARTAMENTO
HAVING QTDE > 19

SELECT D.DS_DEPARTAMENTO DEPARTAMENTO, 
       COUNT(*) QTDE,
       SUM(F.VL_SALARIO) FOLHA
    FROM FUNCIONARIOS f
    JOIN departamentos D 
      ON F.id_departamento = D.id_departamento
   WHERE D.DS_DEPARTAMENTO IN('CONTABILIDADE','FINANCEIRO')
 GROUP BY D.DS_DEPARTAMENTO
HAVING QTDE > 19 AND FOLHA > 138000;

       C.DS_CARGO

    JOIN departamentos D 
      ON F.id_departamento = D.id_departamento
    JOIN cargos c
      ON f.id_cargo = c.id_cargo
