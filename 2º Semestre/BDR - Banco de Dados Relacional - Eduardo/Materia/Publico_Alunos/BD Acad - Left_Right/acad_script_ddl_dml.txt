CREATE DATABASE ACAD;

USE ACAD;

CREATE TABLE Alunos (
    Id_Aluno int PRIMARY KEY,
    id_curso int,
    Nm_Aluno Varchar(30),
    Sg_Genero Char(1),
    Id_UF int);

CREATE TABLE Estados (
    Id_UF int PRIMARY KEY,
    ds_uf char(2),
    Nm_UF Varchar(30)
);

CREATE TABLE Cursos (
    Id_Curso Int PRIMARY KEY,
    Ds_Curso Varchar(40)
);
