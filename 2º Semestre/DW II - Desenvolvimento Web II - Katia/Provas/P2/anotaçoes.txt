```sql
CREATE TABLE produto (
  codigo INT NOT NULL AUTO_INCREMENT,
  nome VARCHAR(100) DEFAULT NULL,
  descricao VARCHAR(100) DEFAULT NULL,
  categoria VARCHAR(100) DEFAULT NULL,
  preco DECIMAL(10,2) DEFAULT NULL,
  observacoes VARCHAR(500) DEFAULT NULL,
```

### 4. Controller (`Controllers/ProdutoController.cs`)

```csharp
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Petshop_MVC.Data;
using Petshop_MVC.Models;

namespace Petshop_MVC.Controllers
{
    public class ProdutoController : Controller
    {
        private readonly AppDbContext _context;

        public ProdutoController(AppDbContext context)
        {
            _context = context;
        }

        [HttpGet]
        public async Task<IActionResult> Index()
        {
            return View(await _context.Produtos.ToListAsync());
        }

        [HttpGet]
        public IActionResult Create()
        {
            return View();
        }

        [HttpPost]
        public async Task<IActionResult> Create([Bind("Codigo, Nome, Descricao, Categoria, Preco, Observacoes")] Produto prod)
        {
            if (ModelState.IsValid)
            {
                _context.Add(prod);
                await _context.SaveChangesAsync();
                return RedirectToAction("Index");
            }
            return View(prod);
        }

        [HttpGet]
        public async Task<IActionResult> Edit(int? id)
        {
            if (id == null) { return NotFound(); }

            var prod = await _context.Produtos.FindAsync(id);
            if (prod == null) { return NotFound(); }

            return View(prod);
        }

        [HttpPost]
        public async Task<IActionResult> Edit(int id, [Bind("Codigo, Nome, Descricao, Categoria, Preco, Observacoes")] Produto prod)
        {
            if (id != prod.Codigo) { return NotFound(); }

            if (ModelState.IsValid)
            {
                _context.Update(prod);
                await _context.SaveChangesAsync();
                return RedirectToAction("Index");
            }
            return View(prod);
        }

        [HttpGet]
        public async Task<IActionResult> Delete(int? id)
        {
            if (id == null) { return NotFound(); }

            var prod = await _context.Produtos.FirstOrDefaultAsync(p => p.Codigo == id);
            if (prod == null) { return NotFound(); }

            return View(prod);
        }

        [HttpPost]
        public async Task<IActionResult> DeleteConfirmed(int? Codigo)
        {
            var prod = await _context.Produtos.FindAsync(Codigo);
            if (prod != null)
            {
                _context.Produtos.Remove(prod);
                await _context.SaveChangesAsync();
            }
            return RedirectToAction("Index");
        }
    }
}
```
